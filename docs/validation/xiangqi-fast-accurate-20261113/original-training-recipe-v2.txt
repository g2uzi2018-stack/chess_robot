import os,json,time,math,random,pathlib
import cv2,numpy as np,torch
from torch import nn
from torch.nn import functional as F
torch.set_num_threads(2);cv2.setNumThreads(4);torch.manual_seed(43);np.random.seed(43)
torch.backends.cudnn.benchmark=True
r=pathlib.Path('/home/kuang/workspace/xiangqi-yolo-training/20261031-yolo26s-seg-v1/dataset')
a=pathlib.Path('/home/kuang/workspace/chess_robot/runs/fast_accurate_20261113_01/classifier-v2');a.mkdir(exist_ok=True)
names=json.loads((r/'data.yaml').read_text())['names']
def crop(im,box):
 c=(box[:2]+box[2:])/2;size=max(box[2:]-box[:2])*1.15
 matrix=np.array([[size/96,0,c[0]-size/2],[0,size/96,c[1]-size/2]],dtype=np.float32)
 return cv2.warpAffine(im,matrix,(96,96),flags=cv2.INTER_LINEAR|cv2.WARP_INVERSE_MAP,borderMode=cv2.BORDER_REPLICATE)
def dataset(split):
 x=[];y=[];ids=[]
 for p in sorted((r/('images/'+split)).iterdir()):
  im=cv2.imread(str(p))
  for idx,line in enumerate((r/('labels/'+split)/p.with_suffix('.txt').name).read_text().splitlines()):
   v=np.array(list(map(float,line.split())));poly=v[1:].reshape(-1,2)*[im.shape[1],im.shape[0]]
   x.append(crop(im,np.r_[poly.min(0),poly.max(0)]));y.append(int(v[0]));ids.append(p.stem+':'+str(idx))
 return torch.tensor(np.stack(x).transpose(0,3,1,2),device='cuda').float()/255,torch.tensor(y,device='cuda'),ids
x,y,ids=dataset('train');vx,vy,vids=dataset('val')
class Net(nn.Module):
 def __init__(self):
  super().__init__();ls=[];n=3
  for c in [24,48,96,128]:
   ls += [nn.Conv2d(n,c,3,padding=1,bias=False),nn.BatchNorm2d(c),nn.SiLU(),nn.Conv2d(c,c,3,padding=1,bias=False),nn.BatchNorm2d(c),nn.SiLU(),nn.MaxPool2d(2)];n=c
  self.features=nn.Sequential(*ls);self.head=nn.Sequential(nn.Flatten(),nn.Linear(128*6*6,128),nn.SiLU(),nn.Dropout(.15),nn.Linear(128,14))
 def forward(self,x):return self.head(self.features(x))
net=Net().cuda();opt=torch.optim.AdamW(net.parameters(),lr=.001,weight_decay=.001);scaler=torch.amp.GradScaler('cuda')
epochs=150;schedule=torch.optim.lr_scheduler.CosineAnnealingLR(opt,epochs,eta_min=.00003)
def aug(z):
 n=len(z);angle=torch.rand(n,device='cuda')*2*math.pi;scale=.9+torch.rand(n,device='cuda')*.2
 th=torch.zeros(n,2,3,device='cuda');th[:,0,0]=torch.cos(angle)*scale;th[:,0,1]=-torch.sin(angle)*scale;th[:,1,0]=torch.sin(angle)*scale;th[:,1,1]=torch.cos(angle)*scale;th[:,:,2]=(torch.rand(n,2,device='cuda')-.5)*.12
 z=F.grid_sample(z,F.affine_grid(th,z.shape,align_corners=False),align_corners=False,padding_mode='border')
 size=random.choice([32,48,64,96])
 z=F.interpolate(F.interpolate(z,size=(size,size),mode='bilinear',align_corners=False),size=(96,96),mode='bilinear',align_corners=False)
 if random.random()<.5:z=F.avg_pool2d(z,3,1,1)
 z=(z*(.55+torch.rand(n,1,1,1,device='cuda')*.9)*(.85+torch.rand(n,3,1,1,device='cuda')*.3)+(torch.rand(n,1,1,1,device='cuda')-.5)*.08).clamp(0,1)
 z=z.pow(.7+torch.rand(n,1,1,1,device='cuda')*.8)
 return z
best=-1;history=[];start=time.monotonic()
for ep in range(epochs):
 net.train();perm=torch.randperm(len(y),device='cuda');losses=[];acc=0
 for ix in perm.split(64):
  opt.zero_grad(set_to_none=True)
  with torch.amp.autocast('cuda'):
   pred=net(aug(x[ix]));loss=F.cross_entropy(pred,y[ix],label_smoothing=.03)
  scaler.scale(loss).backward();scaler.step(opt);scaler.update();losses.append(float(loss));acc+=int((pred.argmax(1)==y[ix]).sum())
 schedule.step();net.eval()
 with torch.no_grad():
  logits=torch.cat([net(z) for z in vx.split(64)]);correct=int((logits.argmax(1)==vy).sum())
 h=dict(epoch=ep+1,train_correct=acc,train_total=len(y),val_correct=correct,val_total=len(vy),loss=float(np.mean(losses)),elapsed_s=time.monotonic()-start);history.append(h)
 if correct>best:
  best=correct;torch.save(dict(state_dict=net.state_dict(),epoch=ep+1,names=names,input_size=96,crop_margin=1.15,train_ids=ids,val_ids=vids),a/'best.pt')
  torch.jit.trace(net,torch.zeros(1,3,96,96,device='cuda')).save(str(a/'classifier.ts'))
 if ep%5==0 or ep==epochs-1:print(json.dumps(h),flush=True)
(a/'training-history.json').write_text(json.dumps(history,indent=2))
print('DONE best validation',best,'/',len(vy),flush=True)
