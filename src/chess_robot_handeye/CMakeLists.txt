cmake_minimum_required(VERSION 3.16)
project(chess_robot_handeye)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(chess_robot_interfaces REQUIRED)
find_package(chess_robot_vision REQUIRED)
find_package(chess_robot_twist REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV 4.5 REQUIRED COMPONENTS core calib3d aruco imgproc imgcodecs)
if(OpenCV_VERSION VERSION_GREATER_EQUAL "4.6")
  message(FATAL_ERROR "This release implements OpenCV 4.5 legacy ChArUco layout. Explicit newer-layout support and validation are required before using OpenCV >= 4.6.")
endif()
find_package(yaml-cpp REQUIRED)
add_library(handeye_core src/core.cpp)
target_include_directories(handeye_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${OpenCV_INCLUDE_DIRS})
target_link_libraries(handeye_core PUBLIC ${OpenCV_LIBS} yaml-cpp)
ament_target_dependencies(handeye_core PUBLIC ament_index_cpp)
add_executable(handeye_node src/node.cpp)
target_link_libraries(handeye_node handeye_core chess_robot_twist::chess_robot_twist)
ament_target_dependencies(handeye_node rclcpp rclcpp_action sensor_msgs std_srvs chess_robot_interfaces chess_robot_vision cv_bridge)
file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/src/core.cpp CORE_SHA)
file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/src/node.cpp NODE_SHA)
file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/include/chess_robot_handeye/core.hpp HEADER_SHA)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/core.cpp src/node.cpp include/chess_robot_handeye/core.hpp)
target_compile_definitions(handeye_node PRIVATE HANDEYE_SOURCE_FINGERPRINT="${CORE_SHA}:${NODE_SHA}:${HEADER_SHA}")
install(TARGETS handeye_node RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS handeye_core EXPORT handeye_core_targets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md PORTING.md DESTINATION share/${PROJECT_NAME})
ament_export_targets(handeye_core_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(OpenCV yaml-cpp ament_index_cpp)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_handeye test/test_core.cpp)
  target_link_libraries(test_handeye handeye_core)
  target_compile_definitions(test_handeye PRIVATE HANDEYE_TEST_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/handeye.example.yaml")
  add_executable(handeye_detector_probe test/detector_probe.cpp)
  target_link_libraries(handeye_detector_probe handeye_core)
  add_executable(handeye_fixture test/fixture.cpp)
  target_link_libraries(handeye_fixture handeye_core chess_robot_twist::chess_robot_twist)
  ament_target_dependencies(handeye_fixture rclcpp rclcpp_action sensor_msgs chess_robot_interfaces chess_robot_vision cv_bridge)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  foreach(case success reject no_motion stale no_vision camera_unhealthy mismatch cancel plan
      cancel_moving_zero_velocity cancel_duplicate_stamp cancel_backward_stamp
      cancel_invalid_feedback cancel_unpaired_feedback)
    add_test(NAME integration_${case} COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/test/integration.py
      --template ${CMAKE_CURRENT_SOURCE_DIR}/config/handeye.example.yaml
      --fixture $<TARGET_FILE:handeye_fixture> --node $<TARGET_FILE:handeye_node>
      --output-parent ${CMAKE_CURRENT_BINARY_DIR}/test_runs --case ${case})
    set_tests_properties(integration_${case} PROPERTIES RUN_SERIAL TRUE TIMEOUT 90)
  endforeach()
endif()
ament_package()
