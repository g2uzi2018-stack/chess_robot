cmake_minimum_required(VERSION 3.16)
project(chess_robot_preview)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(yaml-cpp REQUIRED)
find_package(Boost REQUIRED COMPONENTS system)
find_package(Threads REQUIRED)
add_executable(preview_node src/node.cpp)
ament_target_dependencies(preview_node rclcpp rcl_interfaces sensor_msgs cv_bridge ament_index_cpp)
target_link_libraries(preview_node ${OpenCV_LIBS} yaml-cpp Boost::system Threads::Threads)
target_include_directories(preview_node PRIVATE ${OpenCV_INCLUDE_DIRS})
target_compile_options(preview_node PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS preview_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch web scripts DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
