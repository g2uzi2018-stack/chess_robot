cmake_minimum_required(VERSION 3.16)
project(chess_robot_vision)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(realsense2 REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/FrameInfo.msg" DEPENDENCIES std_msgs)
add_library(camera_calibration STATIC src/calibration.cpp)
target_include_directories(camera_calibration PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> ${OpenCV_INCLUDE_DIRS})
target_link_libraries(camera_calibration PUBLIC ${OpenCV_LIBS} realsense2::realsense2)
target_compile_options(camera_calibration PRIVATE -Wall -Wextra -Wpedantic)
add_executable(vision_node src/node.cpp)
target_link_libraries(vision_node camera_calibration realsense2::realsense2)
# Preserve the external SDK library directory for user-prefix installations.
set_target_properties(vision_node PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
ament_target_dependencies(vision_node rclcpp rcl_interfaces std_msgs sensor_msgs cv_bridge)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(vision_node "${cpp_typesupport_target}")
target_compile_options(vision_node PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS vision_node RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_camera_calibration test/test_calibration.cpp)
  target_link_libraries(test_camera_calibration camera_calibration)
endif()
ament_package()
