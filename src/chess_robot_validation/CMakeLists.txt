cmake_minimum_required(VERSION 3.8)
project(chess_robot_validation)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(kdl_parser REQUIRED)
find_package(urdf REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(nova5_readonly src/readonly.cpp)
add_executable(nova5_transport_check src/transport_check.cpp)
target_link_libraries(nova5_transport_check pthread)
add_executable(nova5_feedback src/feedback.cpp)
ament_target_dependencies(nova5_feedback rclcpp sensor_msgs)
add_executable(nova5_model_check src/model_check.cpp)
ament_target_dependencies(nova5_model_check urdf kdl_parser)
target_link_libraries(nova5_model_check Eigen3::Eigen orocos-kdl)
add_executable(nova5_tf_check src/tf_check.cpp)
ament_target_dependencies(nova5_tf_check rclcpp sensor_msgs tf2_ros)
target_link_libraries(nova5_tf_check Eigen3::Eigen)
install(TARGETS nova5_readonly nova5_feedback nova5_model_check nova5_tf_check nova5_transport_check DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
