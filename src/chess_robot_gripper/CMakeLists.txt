cmake_minimum_required(VERSION 3.16)
project(chess_robot_gripper)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(chess_robot_interfaces REQUIRED)
find_package(Threads REQUIRED)
add_library(gripper_core src/core.cpp src/transport.cpp)
target_include_directories(gripper_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(gripper_core PUBLIC Threads::Threads)
target_compile_options(gripper_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(gripper_node src/node.cpp)
target_link_libraries(gripper_node gripper_core)
ament_target_dependencies(gripper_node rclcpp rclcpp_action rcl_interfaces chess_robot_interfaces)
target_compile_options(gripper_node PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS gripper_node gripper_core RUNTIME DESTINATION lib/${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_gripper test/test_gripper.cpp)
  target_link_libraries(test_gripper gripper_core)
endif()
ament_package()
