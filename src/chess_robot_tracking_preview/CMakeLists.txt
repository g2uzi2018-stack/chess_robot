cmake_minimum_required(VERSION 3.16)
project(chess_robot_tracking_preview)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
find_package(yaml-cpp REQUIRED)
add_executable(tracking_preview src/node.cpp)
ament_target_dependencies(tracking_preview rclcpp sensor_msgs std_msgs cv_bridge)
target_link_libraries(tracking_preview yaml-cpp ${OpenCV_LIBS})
target_compile_options(tracking_preview PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS tracking_preview RUNTIME DESTINATION lib/${PROJECT_NAME})
ament_package()
