cmake_minimum_required(VERSION 3.16)
project(chess_arm_maintenance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
add_library(arm_tools src/operator.cpp)
target_include_directories(arm_tools PUBLIC include)
target_link_libraries(arm_tools PUBLIC yaml-cpp Threads::Threads)
target_compile_options(arm_tools PRIVATE -Wall -Wextra -Wpedantic)
add_executable(arm_operator src/main.cpp)
target_link_libraries(arm_operator PRIVATE arm_tools)
target_compile_options(arm_operator PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  add_executable(test_operator test/test_operator.cpp)
  target_link_libraries(test_operator PRIVATE arm_tools GTest::gtest_main)
  add_test(NAME maintenance COMMAND test_operator)
  set_tests_properties(maintenance PROPERTIES TIMEOUT 20)
endif()
